cmake_minimum_required(VERSION 3.22.1)
project(ktlottie C)

if(ANDROID)
    if(ANDROID_ABI STREQUAL "arm64-v8a")
        set(RUST_TARGET aarch64-linux-android)
    elseif(ANDROID_ABI STREQUAL "armeabi-v7a")
        set(RUST_TARGET armv7-linux-androideabi)
    elseif(ANDROID_ABI STREQUAL "x86_64")
        set(RUST_TARGET x86_64-linux-android)
    elseif(ANDROID_ABI STREQUAL "x86")
        set(RUST_TARGET i686-linux-android)
    else()
        message(FATAL_ERROR "no rust target for ${ANDROID_ABI}")
    endif()
    set(TLOTTIE_LIB ${TLOTTIE_TARGET_DIR}/${RUST_TARGET}/release/libtlottie.a)
else()
    set(TLOTTIE_LIB ${TLOTTIE_TARGET_DIR}/release/libtlottie.a)
endif()

add_library(tlottie STATIC IMPORTED)
set_target_properties(tlottie PROPERTIES IMPORTED_LOCATION ${TLOTTIE_LIB})

add_library(ktlottie SHARED tlottie_jni.c)
target_include_directories(ktlottie PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../rust/tlottie/include)
target_compile_options(ktlottie PRIVATE -O2 -fvisibility=hidden -Wall -Wextra -Werror)
target_link_libraries(ktlottie PRIVATE tlottie m dl)

if(ANDROID)
    target_link_libraries(ktlottie PRIVATE log)
    target_link_options(ktlottie PRIVATE -Wl,-z,max-page-size=16384 -Wl,--gc-sections -Wl,--exclude-libs,ALL)
else()
    find_package(JNI REQUIRED)
    target_include_directories(ktlottie PRIVATE ${JNI_INCLUDE_DIRS})
    find_package(Threads REQUIRED)
    target_link_libraries(ktlottie PRIVATE Threads::Threads)
endif()
